// --- ゲーム変数 ---
let currentTurn = 0;
let stats = { maou: 0, syousai: 0, sinchou: 0, kisaku: 0, danger: 0 };

// --- 画面更新処理 ---
function loadTurn() {
  if (currentTurn >= gameData.length) {
    showEnding();
    return;
  }

  const data = gameData[currentTurn];
  document.getElementById("turn-title").innerText = `【第${data.turn}ターン】 ${data.title}`;
  document.getElementById("question-text").innerText = data.question;
  document.getElementById("result-container").style.display = "none";

  const optionsBox = document.getElementById("options-container");
  optionsBox.innerHTML = "";
  optionsBox.style.display = "flex";

  data.options.forEach(opt => {
    const btn = document.createElement("button");
    btn.className = "btn";
    btn.innerText = opt.text;
    btn.onclick = () => selectOption(opt);
    optionsBox.appendChild(btn);
  });
}

// --- 選択肢クリック時 ---
function selectOption(opt) {
  stats[opt.type]++;
  stats.danger += opt.danger;
  updateStatus();

  // 本能寺危険度100%での即死判定
  if (stats.danger >= 100) {
    showGameOver();
    return;
  }

  document.getElementById("options-container").style.display = "none";
  document.getElementById("result-text").innerText = opt.result;
  document.getElementById("result-container").style.display = "block";
}

function nextTurn() {
  currentTurn++;
  loadTurn();
}

function updateStatus() {
  document.getElementById("pt-maou").innerText = stats.maou;
  document.getElementById("pt-syousai").innerText = stats.syousai;
  document.getElementById("pt-sinchou").innerText = stats.sinchou;
  document.getElementById("pt-kisaku").innerText = stats.kisaku;
  document.getElementById("danger-degree").innerText = `${stats.danger}%`;
}

// --- 判定・エンディング処理 ---
function showEnding() {
  document.getElementById("options-container").style.display = "none";
  document.getElementById("result-container").style.display = "none";
  document.getElementById("turn-title").innerText = "【プロトタイプ体験版 終了】";

  let highestPt = Object.keys(stats).reduce((a, b) => b === 'danger' ? a : (stats[a] > stats[b] ? a : b));
  let endTitle = "";
  let endDesc = "";

  if (highestPt === "maou") {
    endTitle = "【魔王覚醒エンド】";
    endDesc = "史実通りの武力と覇気で着実に天下統一へと突き進んでいます！";
  } else if (highestPt === "syousai") {
    endTitle = "【大財閥エンド】";
    endDesc = "莫大な資金力を背景に、戦わずして日本経済を裏から支配しました！";
  } else if (highestPt === "sinchou") {
    endTitle = "【堅実長寿エンド】";
    endDesc = "無駄なリスクを徹底排除し、安全に領地を広げています！";
  } else {
    endTitle = "【カオスギャグエンド】";
    endDesc = "常識破りの奇行で周囲を困惑させつつ、謎の勢いで歴史を塗り替えています！";
  }

  document.getElementById("question-text").innerHTML = `
    <h3>${endTitle}</h3>
    <p>${endDesc}</p>
    <p>3ターンのプレイありがとうございました！プロトタイプの完成です。</p>
    <button class="btn next-btn" onclick="location.reload()">最初からやり直す</button>
  `;
}

function showGameOver() {
  document.getElementById("options-container").style.display = "none";
  document.getElementById("result-container").style.display = "none";
  document.getElementById("turn-title").innerText = "【GAME OVER】";
  document.getElementById("question-text").innerHTML = `
    <h3 style="color:#ff4d4d;">【本能寺の変（早期勃発）】</h3>
    <p>本能寺危険度が100%に達しました！明智光秀のストレスが限界突破し、謀反を起こされました……。</p>
    <button class="btn next-btn" onclick="location.reload()">もう一度挑戦する</button>
  `;
}
